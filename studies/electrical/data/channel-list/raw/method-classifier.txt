def scope_title(t):
    t=t or ""
    if EBIKE.search(t): return "in"
    t=STRIP.sub(" ", t)
    i=any(r.search(t) for r in INR); o=any(r.search(t) for r in OUTR)
    if i and not o: return "in"
    if i and o: return "in" if any(r.search(t) for r in STRONG) and not re.search(r"solar|power station|ecoflow|jackery|bluetti|prepp|plumb|hvac|furnace|heat pump|air condition|water heater|burglar|security", t, re.I) else "out"
    return "out"
def secs(d):
    if not d: return None
    p=[int(x) for x in d.split(":")]
    s=0
    for x in p: s=s*60+x
    return s

def nonenglish(t):
    t=t or ""
    if NONLATIN.search(t): return True
    if HING.search(t): return True
    return len(FOREIGN.findall(t))>=2
